cmake_minimum_required(VERSION 3.19)
project(rocketperf LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/project/modules.json" MODULE_REGISTRY)
function(module_sources output module_id)
    string(JSON module_count LENGTH "${MODULE_REGISTRY}" modules)
    math(EXPR last_module "${module_count} - 1")
    set(sources)
    foreach(index RANGE 0 ${last_module})
        string(JSON candidate GET "${MODULE_REGISTRY}" modules ${index} id)
        if("${candidate}" STREQUAL "${module_id}")
            string(JSON count LENGTH "${MODULE_REGISTRY}" modules ${index} sources)
            if(count GREATER 0)
                math(EXPR last "${count} - 1")
                foreach(item RANGE 0 ${last})
                    string(JSON source GET "${MODULE_REGISTRY}" modules ${index} sources ${item})
                    list(APPEND sources "${source}")
                endforeach()
            endif()
        endif()
    endforeach()
    set(${output} ${sources} PARENT_SCOPE)
endfunction()
module_sources(CORE_SOURCES core)
module_sources(NOZZLE_SOURCES nozzle)
module_sources(THERMO_SOURCES thermo)
module_sources(ADAPTER_SOURCES adapters)
module_sources(CLI_SOURCES cli)
add_library(rocketperf_core STATIC ${CORE_SOURCES} ${NOZZLE_SOURCES} ${THERMO_SOURCES})
target_include_directories(rocketperf_core PUBLIC include)
if(NOT MSVC)
    target_link_libraries(rocketperf_core PUBLIC m)
endif()

add_executable(rocketperf ${CLI_SOURCES} ${ADAPTER_SOURCES})
target_include_directories(rocketperf PRIVATE src/adapters)
target_link_libraries(rocketperf PRIVATE rocketperf_core)
if(MINGW)
    target_link_options(rocketperf PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.c)
    target_link_libraries(test_core PRIVATE rocketperf_core)
    add_test(NAME core COMMAND test_core)
    add_executable(test_thermo tests/test_thermo.c)
    target_link_libraries(test_thermo PRIVATE rocketperf_core)
    add_test(NAME thermo COMMAND test_thermo)
    add_executable(test_adapters tests/test_adapters.c ${ADAPTER_SOURCES})
    target_include_directories(test_adapters PRIVATE src/adapters)
    target_link_libraries(test_adapters PRIVATE rocketperf_core)
    add_test(NAME adapters COMMAND test_adapters)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py"
            --binary "$<TARGET_FILE:rocketperf>")
    endif()
endif()

foreach(target rocketperf_core rocketperf test_core test_adapters test_thermo)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /fp:precise)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror
                -Wconversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes -fno-common)
        endif()
    endif()
endforeach()
